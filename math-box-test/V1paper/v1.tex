\documentclass{article}

\usepackage{zref-savepos}

% Box used to measure each equation
\newsavebox{\mathbox}

% Output file
\newwrite\mathcoords
\immediate\openout\mathcoords=mathcoords.csv

% Unique ID for every math expression
\newcounter{mathid}

% CSV header
\immediate\write\mathcoords{id,page,x,y,width,height,depth}

\newcommand{\recordmath}[1]{%
    \stepcounter{mathid}%

    % Put the equation into a temporary box.
    % This lets TeX tell us width / height / depth.
    \sbox{\mathbox}{$#1$}%

    % Give this position a unique name
    \edef\mathlabel{math-\themathid}%

    % Record the current position in the PDF
    \zsavepos{\mathlabel}%

    % Write annotation information
    %
    % x/y come from the PREVIOUS LaTeX compilation,
    % which is why we compile twice.
    \immediate\write\mathcoords{%
        \themathid,
        \thepage,
        \zposx{\mathlabel},
        \zposy{\mathlabel},
        \number\wd\mathbox,
        \number\ht\mathbox,
        \number\dp\mathbox
    }%

    % Actually display the equation normally
    \usebox{\mathbox}%
}

\AtEndDocument{%
    \immediate\closeout\mathcoords
}




\begin{document}

Newton's second law is \recordmath{F = ma}.

Einstein's famous equation is
\recordmath{E = mc^2}.

Here is another equation:
\recordmath{x^2 + y^2 = z^2}.

\end{document}
