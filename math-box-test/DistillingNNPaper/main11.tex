\newcommand{\recordmath}[1]{%
    % amsmath measures aligned equations before typesetting them. During that
    % pass, render normally without incrementing the ID or writing a CSV row.
    \ifmeasuring@
        #1%
    \else
        \stepcounter{mathid}%
        \sbox{\mathbox}{$#1$}%
        \edef\mathlabel{math-\themathid}%
        \leavevmode
        \zsavepos{\mathlabel}%
        \immediate\write\mathcoords{%
            \themathid,
            \zref@extractdefault{\mathlabel}{abspage}{0},
            \zposx{\mathlabel},
            \zposy{\mathlabel},
            \number\wd\mathbox,
            \number\ht\mathbox,
            \number\dp\mathbox
        }%
        \usebox{\mathbox}%
    \fi
}
\makeatother
